## Tests

- Los tests de arnés que hoy acceden a privados del núcleo por cast se mueven al test del módulo nuevo, con un puerto falso tipado con `satisfies` y sin `as never` ni `as unknown as` contra el módulo.
- Los tests de cableado solo usan la fachada pública.
- Contra el verde por casts: mismos títulos y número de `expect(` antes y después, sabotaje por paso y recuento con `private-access.mjs`.

## Censo de observabilidad

- El id de una frontera es un sha1 de ruta, tipo, registro, cadena de ancestros `Clase.método`, texto del nodo sin comentarios y número de ocurrencia. No incluye la línea. Mover de fichero o de clase cambia siempre el id.
- La evidencia depende de `scope`, `behavior` y del `callee` observado; por eso se conservan los nombres de método y campo.
- Renombrado: `census-remap.mjs` copia lo que conserva id, empareja lo movido por tipo, registro, cadena sin nombre de clase y texto, y reaplica la decisión con `reviewActionBoundaryCensus`. Lo ambiguo queda sin revisar y se lista. Probado en una copia: 6 métodos movidos, 5 fronteras recolocadas, `verify` 0; con el texto alterado a propósito, 1 sin pareja y `verify` rojo. No está probado cuando el cableado pasa a funciones `assemble*`.

let cls;

const members = [];
for (const m of cls.members) {
  const name = m.name ? m.name.getText(sf) : (ts.isConstructorDeclaration(m) ? 'constructor' : '?');
  const mods = (ts.getModifiers(m) || []).map((x) => x.getText(sf));
  const vis = mods.includes('private') ? 'private' : mods.includes('protected') ? 'protected' : name.startsWith('#') ? 'hash' : 'public';
  const kind = ts.isPropertyDeclaration(m) ? 'field' : ts.isMethodDeclaration(m) ? 'method' : ts.isGetAccessor(m) ? 'get' : ts.isSetAccessor(m) ? 'set' : ts.isConstructorDeclaration(m) ? 'ctor' : 'other';

  const refs = new Set();
  const visit = (n) => {
    if (ts.isPropertyAccessExpression(n) && n.expression.kind === ts.SyntaxKind.ThisKeyword) refs.add(n.name.getText(sf));
    ts.forEachChild(n, visit);
  };
  visit(m);
  members.push({ name, vis, kind: isArrowField ? 'arrow' : kind, start: line(m.getStart(sf)), end: line(m.end), loc: line(m.end) - line(m.getStart(sf)) + 1, refs: [...refs] });
}

const by = (k) => members.filter((m) => m.kind === k);
for (const k of ['field', 'arrow', 'method', 'get', 'set', 'ctor']) {
  const l = by(k); console.log(k, l.length, 'public', l.filter((m) => m.vis === 'public').length, 'private', l.filter((m) => m.vis !== 'public').length);

for (const m of members) console.log(`${m.start}-${m.end}\t${m.loc}\t${m.kind}\t${m.vis}\t${m.name}`);

### `census-remap.mjs`

```js
// Uso: node census-remap.mjs <root> --old-root=<dir con el arbol ANTERIOR (git worktree o git archive)> [--write]
// Recoloca las decisiones del censo de observabilidad cuando un lote MUEVE codigo de fichero sin cambiarlo.
// 1) Frontera con el mismo id en el mismo fichero: se copia la entrada vieja tal cual (solo cambia el localizador).
// 2) Frontera nueva: se busca en el arbol viejo una frontera decidida con la misma clave
//    kind | registration | cadena de declaraciones SIN nombres de clase | texto normalizado, cuyo id ya no exista.
//    Si hay exactamente una, se reaplica su decision con reviewActionBoundaryCensus (la evidencia la recalcula el AST).
// 3) Lo demas queda `unreviewed` y se lista: eso lo revisa una persona.
// Sin --write solo informa. Con --write reescribe scripts/action-observability-baseline.json con tabuladores.
import { readFileSync, writeFileSync } from 'node:fs';
import { join, resolve } from 'node:path';
import { pathToFileURL } from 'node:url';
const root = resolve(process.argv[2]);
const oldRoot = resolve(process.argv.find((a) => a.startsWith('--old-root=')).slice('--old-root='.length));
const census = await import(pathToFileURL(join(root, 'scripts/action-observability-census.mjs')).href);
const ts = (await import(pathToFileURL(join(root, 'node_modules/typescript/lib/typescript.js')).href)).default;
const BASELINE = 'scripts/action-observability-baseline.json';
const oldBaseline = JSON.parse(readFileSync(join(oldRoot, BASELINE), 'utf8'));
const loc = (b) => `${b.line}:${b.column}-${b.endLine}:${b.endColumn}`;
const astCache = new Map();
function keyAt(dir, path, kind, registration, locator) {
  const file = join(dir, path);
  if (!astCache.has(file)) {
    const text = readFileSync(file, 'utf8');
    const sf = ts.createSourceFile(path, text, ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);
    const nodes = new Map();
    const lc = (p) => { const x = sf.getLineAndCharacterOfPosition(p); return `${x.line + 1}:${x.character + 1}`; };
    const visit = (n) => { nodes.set(`${lc(n.getStart(sf))}-${lc(n.end)}|${ts.SyntaxKind[n.kind]}`, n); ts.forEachChild(n, visit); };
    visit(sf); astCache.set(file, { sf, nodes });
  }
  const { sf, nodes } = astCache.get(file);
  const want = { catch_clause: 'CatchClause', void_expression: 'VoidExpression', promise_catch: 'CallExpression', callback_registration: 'CallExpression' }[kind];
  const node = nodes.get(`${locator}|${want}`);
  if (node === undefined) return null;
  const names = [];
  for (let c = node.parent; c; c = c.parent) {
    if (ts.isMethodDeclaration(c) || ts.isFunctionDeclaration(c) || ts.isGetAccessor(c) || ts.isSetAccessor(c) || ts.isConstructorDeclaration(c)) names.unshift(ts.isConstructorDeclaration(c) ? 'constructor' : c.name?.getText(sf) ?? '<anonymous>');
    else if ((ts.isArrowFunction(c) || ts.isFunctionExpression(c)) && (ts.isVariableDeclaration(c.parent) || ts.isPropertyDeclaration(c.parent) || ts.isPropertyAssignment(c.parent))) names.unshift(c.parent.name.getText(sf));
  }
  const text = node.getText(sf).replace(/\/\*[\s\S]*?\*\//gu, '').replace(/\/\/[^\n]*/gu, '').trim();
  return JSON.stringify([kind, registration ?? '', names.join('.'), text]);
}
const fresh = census.collectActionBoundaryCensus(root);
const freshIds = new Set(Object.entries(fresh.files).flatMap(([p, f]) => f.boundaries.map((b) => `${p}:${b.id}`)));
const movedCandidates = new Map();
for (const [path, file] of Object.entries(oldBaseline.files)) for (const b of file.boundaries) {
  if (b.classification === 'unreviewed' || freshIds.has(`${path}:${b.id}`)) continue;
  const key = keyAt(oldRoot, path, b.kind, b.registration, loc(b));
  if (key === null) { console.log(`WARN old boundary not found in old tree: ${path} ${loc(b)}`); continue; }
  movedCandidates.set(key, [...(movedCandidates.get(key) ?? []), { path, b }]);
}
const report = { kept: 0, moved: [], ambiguous: [], unmatched: [] };
const reviewed = census.reviewActionBoundaryCensus(root, (c) => {
  const oldFile = oldBaseline.files[c.path];
  const key = keyAt(root, c.path, c.kind, c.registration, c.locator);
  const hits = movedCandidates.get(key) ?? [];
  if (hits.length === 0) return undefined;
  if (hits.length > 1) { report.ambiguous.push(`${c.path} ${c.locator} <- ${hits.map((h) => `${h.path} ${loc(h.b)}`).join(' | ')}`); return undefined; }
  const { path, b } = hits[0];
  report.moved.push(`${path} ${loc(b)} -> ${c.path} ${c.locator} [${b.classification}]`);
  return b.classification === 'observed' ? { classification: 'observed', callee: b.evidence.callee }
    : { classification: 'allowlisted', justification: b.evidence.justification };
});
for (const [path, file] of Object.entries(reviewed.files)) {
  const oldById = new Map((oldBaseline.files[path]?.boundaries ?? []).map((b) => [b.id, b]));
  file.boundaries = file.boundaries.map((b) => {
    const old = oldById.get(b.id);
    if (old === undefined) { if (b.classification === 'unreviewed') report.unmatched.push(`${path} ${loc(b)} ${b.kind}`); return b; }
    report.kept += 1;
    return { ...old, line: b.line, column: b.column, endLine: b.endLine, endColumn: b.endColumn };
  });
  if (oldBaseline.files[path]) { file.classification = oldBaseline.files[path].classification; file.reason = oldBaseline.files[path].reason; }
}
console.log(JSON.stringify({ kept: report.kept, moved: report.moved.length, ambiguous: report.ambiguous.length, unmatched: report.unmatched.length }));
for (const k of ['moved', 'ambiguous', 'unmatched']) for (const line of report[k]) console.log(`${k}: ${line}`);
if (process.argv.includes('--write')) {
  writeFileSync(join(root, BASELINE), `${JSON.stringify(reviewed, null, '\t')}\n`);
  const findings = census.verifyActionBoundaryCensus(root);
  console.log(`verify after write: ${findings.length} finding(s)`);
  for (const f of findings) console.log(`  ${JSON.stringify(f)}`);
}

```
